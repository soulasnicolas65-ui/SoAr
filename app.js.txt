let carrito = [];
let total = 0;

function renderizarProductos(listaProductos = productos) {
    const contenedor = document.getElementById('contenedor-productos');
    contenedor.innerHTML = '';

    if (listaProductos.length === 0) {
        contenedor.innerHTML = '<p style="grid-column: 1 / -1; text-align: center; color: #666;">No encontramos productos con ese nombre.</p>';
        return;
    }

    listaProductos.forEach(prod => {
        let opcionesHTML = '';
        prod.opciones.forEach((opcion, index) => {
            opcionesHTML += `<option value="${index}">${opcion.peso} - $${opcion.precio}</option>`;
        });

        const tarjeta = `
            <div class="producto-card">
                <div class="producto-img">
                    <img src="${prod.imagen}" alt="${prod.nombre}">
                </div>
                <h3>${prod.nombre}</h3>
                <select id="select-prod-${prod.id}">
                    ${opcionesHTML}
                </select>
                <button class="btn-agregar" onclick="agregarAlCarrito(${prod.id})">Agregar al pedido</button>
            </div>
        `;
        contenedor.innerHTML += tarjeta;
    });
}

function filtrarProductos() {
    const texto = document.getElementById('buscador').value.toLowerCase();
    const productosFiltrados = productos.filter(prod => 
        prod.nombre.toLowerCase().includes(texto)
    );
    renderizarProductos(productosFiltrados);
}

function agregarAlCarrito(idProducto) {
    const producto = productos.find(p => p.id === idProducto);
    const select = document.getElementById(`select-prod-${idProducto}`);
    const indiceOpcion = select.value;
    const opcionSeleccionada = producto.opciones[indiceOpcion];

    carrito.push({
        nombre: producto.nombre,
        peso: opcionSeleccionada.peso,
        precio: opcionSeleccionada.precio
    });

    total += opcionSeleccionada.precio;
    actualizarVistaCarrito();
}

function actualizarVistaCarrito() {
    const lista = document.getElementById('lista-carrito');
    lista.innerHTML = ''; 

    if (carrito.length === 0) {
        lista.innerHTML = '<li class="carrito-vacio">Tu carrito está vacío</li>';
        document.getElementById('total-precio').innerText = '0';
        return;
    }

    carrito.forEach(item => {
        const li = document.createElement('li');
        li.innerHTML = `<span>${item.nombre} (${item.peso})</span> <span>$${item.precio}</span>`;
        lista.appendChild(li);
    });

    document.getElementById('total-precio').innerText = total;
}

function enviarWhatsApp() {
    if (carrito.length === 0) {
        alert("Agrega productos al pedido primero.");
        return;
    }

    const numero = "5491100000000"; // Tu número aquí
    let mensaje = "¡Hola SoAr Tienda Natural! Quería hacer el siguiente pedido:%0A%0A";
    
    carrito.forEach(item => {
        mensaje += `🔸 ${item.nombre} - ${item.peso} ($${item.precio})%0A`;
    });
    
    mensaje += `%0A*Total: $${total}*%0A%0A¡Muchas gracias!`;
    window.open(`https://wa.me/${numero}?text=${mensaje}`, '_blank');
}

// Iniciar al abrir la página
renderizarProductos();
